## 1. `addPointScaleQuestion()` and `addMultiScaleQuestion()` — full current code

From [page-objects/survey/surveyBuilder.page.ts](page-objects/survey/surveyBuilder.page.ts):

```ts
  /**
   * TC_BS_08 — Point Scale. Row/column add buttons are UNCONFIRMED (see
   * class-level comment #3); only pass extraRows/extraColumns if you have
   * verified "+ Row"/"+ Column" exist for this question type.
   */
  async addPointScaleQuestion(
    questionIndex: number,
    questionText: string,
    rows: string[],
    columns: string[]
  ): Promise<void> {
    try {
      await this.clickQuestionTypeResource('questionTypePointScale');
      await this.fillQuestionText(questionIndex, questionText);
      for (let i = 0; i < rows.length; i++) {
        if (i >= 2) {
          await this.addRowToQuestion(); // UNCONFIRMED for Point Scale
        }
        await this.fillLabeledField(`Row ${i + 1}`, rows[i]);
      }
      for (let i = 0; i < columns.length; i++) {
        if (i >= 2) {
          await this.addColumnToQuestion(); // UNCONFIRMED for Point Scale
        }
        await this.fillLabeledField(`Column ${i + 1}`, columns[i]);
      }
    } catch (error) {
      throw new Error(`Failed to add Point Scale question: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** TC_BS_09 — Multi Scale, including add-row/add-column (confirmed real per Codegen + TC_BS_09). */
  async addMultiScaleQuestion(
    questionIndex: number,
    questionText: string,
    rows: string[],
    columns: string[]
  ): Promise<void> {
    try {
      await this.clickQuestionTypeResource('questionTypeMultiScale');
      await this.fillQuestionText(questionIndex, questionText);
      for (let i = 0; i < rows.length; i++) {
        if (i >= 2) {
          await this.addRowToQuestion();
        }
        await this.fillLabeledField(`Row ${i + 1}`, rows[i]);
      }
      for (let i = 0; i < columns.length; i++) {
        if (i >= 2) {
          await this.addColumnToQuestion();
        }
        await this.fillLabeledField(`Column ${i + 1}`, columns[i]);
      }
    } catch (error) {
      throw new Error(`Failed to add Multi Scale question: ${error instanceof Error ? error.message : error}`);
    }
  }
```

Both methods exist and are implemented. Neither currently has a finalize/"Add question" click at
the end (unlike `addMultipleChoiceQuestion()`, `addRatingQuestion()`, and `addNPSQuestion()`, which
all had that click added in earlier fixes this session).
