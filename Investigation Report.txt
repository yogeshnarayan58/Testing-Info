```ts
import { Reporter, Suite, TestCase, TestResult } from '@playwright/test/reporter';
import { FullConfig } from "@playwright/test";
import * as path from 'path';
import { config } from './config';
import { PostgresActions, CustomReporterOptions, CommonReporterUtils } from "@gfk/gta";


class GrafanaPlaywrightReporter implements Reporter {

    private areaId: number | undefined;
    private domainId: number | undefined;
    private teamId: number | undefined;
    private options: CustomReporterOptions;
    private projectId: number | undefined;
    private environmentName: string;
    private commonReporterUtils: CommonReporterUtils | undefined;
    private postgresActions: PostgresActions;
    private previousTestCaseId: string | null | undefined;
    private resultID: number | undefined;

    constructor(options: CustomReporterOptions) {
        this.options = options;
        console.log(`Custom reporter - Custom options received`);
        this.environmentName = config.ENV;
        this.postgresActions = new PostgresActions(config.postgresAPIUrl);
    }


    async initialize(): Promise<void> {
        if (!this.commonReporterUtils) {
            throw new Error('CommonReporterUtils not initialized. Call onBegin first.');
        }
        const { areaId, domainId, teamId, projectId } = await this.commonReporterUtils.getOrCreateIds();
        this.areaId = areaId;
        this.domainId = domainId;
        this.teamId = teamId;
        this.projectId = projectId;
    }

    async onBegin(fullConfig: FullConfig, suite: Suite) {
        console.log(`Custom reporter - Starting the run with ${suite.allTests().length} tests`);
        this.commonReporterUtils = new CommonReporterUtils(this.options, this.environmentName, config.postgresAPIUrl);
        await this.initialize();
    }

    async onTestEnd(testCase: TestCase, result: TestResult): Promise<void> {
        try {
            if (this.projectId) {
                const isRetry = this.previousTestCaseId === testCase.id;
                const suiteName = path.basename(testCase.parent?.title, '.ts');
                // testCaseId: To read from test case tags/annotation
                // testCycleName: process.env.JIRA_CYCLE_NAME || '',
                // releaseVersion: process.env.VERSION_NAME || '',
                this.resultID = await this.postgresActions.createOrUpdateTestResults(
                    {
                        suiteName: suiteName,
                        testcaseName: testCase.title,
                        status: result.status,
                        executionTime: result.startTime,
                        executionDuration: result.duration,
                        envName: this.environmentName,
                        projectId: this.projectId,
                    },
                    isRetry,
                    this.resultID
                );
                console.log('Custom reporter: - Test result created:', this.resultID);
                this.previousTestCaseId = testCase.id;
            } else {
                console.error('Custom reporter - Project ID is not available');
            }
        } catch (error) {
            console.error('Custom reporter - Error creating test result:', error);
        }
    }
}

export default GrafanaPlaywrightReporter;
```

## 4. Does it send data to an external API? What's the endpoint pattern?

**Yes** — it sends data to an external HTTP API (a "postgres-api" service, presumably backing a
Grafana dashboard), not directly to Grafana's own API. The reporter itself never calls `fetch`/`axios`
directly — all network calls are delegated to two classes imported from the private `@gfk/gta` package:

- `PostgresActions` (constructed with `config.postgresAPIUrl`) — `.createOrUpdateTestResults(...)` is
  called in `onTestEnd()` for every test, sending:
  ```ts
  {
    suiteName,        // spec file name (path.basename of parent suite title)
    testcaseName,     // test title
    status,           // 'passed' | 'failed' | 'timedOut' | 'skipped' | ...
    executionTime,    // result.startTime
    executionDuration,// result.duration
    envName,          // config.ENV
    projectId,        // resolved numeric ID
  }
  ```
  plus an `isRetry` flag and the previous `resultID` (for update-vs-create semantics).
- `CommonReporterUtils` (constructed with `options`, `environmentName`, `config.postgresAPIUrl`) —
  `.getOrCreateIds()` is called once in `onBegin()`/`initialize()` to resolve
  `areaName`/`domainName`/`teamName`/`projectName` strings into numeric
  `areaId`/`domainId`/`teamId`/`projectId` (presumably an upsert-by-name lookup against the API).

**Endpoint pattern** (constructed from `config.postgresAPIUrl` base + implied REST resource paths,
inferred from the method names since `@gfk/gta`'s internals are a private package not present in
`node_modules` here and could not be inspected directly):

```
Base URL:  <POSTGRES_API_URL>            e.g. https://postgres-api.qa.in.gfk.com/report-postgres/api/v1
Calls made (inferred from PostgresActions/CommonReporterUtils method names, not the literal paths):
  - getOrCreateIds()             → some "area/domain/team/project" lookup-or-create endpoint(s) under the base URL
  - createOrUpdateTestResults()  → some "test-results" create/update endpoint under the base URL
```
No API key, token, or auth header is visible anywhere in `grafana-reporter.ts`, `config.ts`, or the
`.env.example` Grafana section — only the base URL and the four naming/grouping strings
(area/domain/team/project) are passed. If `PostgresActions`/`CommonReporterUtils` do send credentials,
they are handled entirely inside the private `@gfk/gta` package (not visible in this workspace).
