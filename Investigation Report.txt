  /**
   * Sets the Madlib selections by iterating through each folder and applying its corresponding dimension selections.
   *
   * @param dimensionSelection - An object containing folder names as keys and their corresponding selections as values.
   */
  public async setMadlibSelection(
    dimensionSelection: {
      selections: Record<string, any>;
    },
    searchReq = true
  ): Promise<void> {

    const folderEntries = Object.entries(dimensionSelection.selections);
    if (!folderEntries.length) {
      return;
    }
    // Clear default/existing state once before processing all folders.
    await this.checkSelectionGroupExists();

    // Loop through each folder and its associated selections.
    for (let index = 0; index < folderEntries.length; index++) {
      const [folderName, selections] = folderEntries[index];
      if (index > 0) { // For second+ folders, create an additional selection group instead of replacing previous selections.
        await this.clickAddIcon();
      }
      await this.setMadlibFolder(folderName, searchReq); // Select the folder
      await this.webActionHandler
        .getWebElement(
          this.resourceManager.getResource("dsSelectionsViewPort", DS_COMPONENT_NAME)
        )
        .waitFor(MAX_TEST_TIMEOUT, "visible");
      await this.setDimensionSelection(selections, searchReq); // Apply the dimension selections within the folder
    }
  }

  /**
   * Opens a Madlib folder by searching for its name and selecting it.
   * Logs an error if the operation fails.
   *
   * @param folderName - The name of the folder to search and select.
   */
  public async setMadlibFolder(
    folderName: string,
    searchReq = true
  ): Promise<void> {
    try {
      if (searchReq) {
         // Get the search icon element
        const searchIcon = this.webActionHandler.getWebElement(
          this.resourceManager.getResource("searchIcon", DS_COMPONENT_NAME)
        );
      // Check if the search icon is visible
        const isVisible = await searchIcon.isVisible();
        if (isVisible) {
          // Click on the search icon to activate the search input
          await this.clickSearchIcon();
          // Enter the folder name into the search field
          await this.enterTextInSearchField(folderName);
        } else {
          // If the search icon is not visible, directly enter the folder name
          // Directly select the folder without searching
          await this.webActionHandler.delay(SMALL_TIMEOUT);
          await this.enterTextInSearchField(folderName);
        }
        // Select the folder from the search results
        await this.selectFolder(folderName);
      } else {
        // Search not required in new UI flows; select directly from visible list.
        await this.selectFolder(folderName);
      }
      console.log(`Folder '${folderName}' successfully selected.`);
    } catch (error) {
      console.error(
        `❌ Failed to select Madlib folder '${folderName}':`,
        error
      );
      throw error;
    }
  }

  /**
   * Waits for madlib stability after pivot changes by monitoring text content stability
   * Ensures that madlib changes at least once, otherwise waits full timeout duration
   * @param {number} timeout - Maximum wait time in milliseconds (default: 10 seconds)
   * @param {number} stabilityDuration - Duration to wait for stability in milliseconds (default: 5 seconds)  
   * @returns Promise<void>
   */
  // TODO: This is a workaround solution. In future, implement network call stability monitoring
  // TODO: instead of relying on UI text changes for detecting pivot switch completion
  public async waitForMadlibStability(timeout: number = 2 * SMALL_TIMEOUT, stabilityDuration: number = SMALL_TIMEOUT): Promise<void> {
    const startTime = Date.now();
    let lastMadlibText = '';
    let stableStartTime = 0;
    let hasDetectedChange = false; // Track if madlib has changed at least once
    
    console.log('Waiting for madlib stability after pivot change...');
    
    try {
      while (Date.now() - startTime < timeout) {
        // Get current madlib text
        const currentMadlibText = await this.getMadlibText();
        
        // Check if madlib text has stabilized
        if (currentMadlibText === lastMadlibText && currentMadlibText.length > 0) {
          // Text is stable - but only exit if we've detected at least one change
          if (hasDetectedChange) {
            if (stableStartTime === 0) {
              stableStartTime = Date.now();
            } else if (Date.now() - stableStartTime >= stabilityDuration) {
              console.log('Madlib text has stabilized after pivot change (change detected)');
              return;
            }
          }
          // If no change detected yet, continue waiting
        } else {
          // Text changed - mark that we've detected a change and reset stability timer
          if (lastMadlibText !== '') {
            hasDetectedChange = true;
            console.log('Madlib text change detected after pivot switch');
            console.log('currentMadlibText:', currentMadlibText);
          }
          lastMadlibText = currentMadlibText;
          stableStartTime = 0;
        }
        
        await this.waitForTimeout(POLLING_INTERVAL); // Check every 300ms
      }
      
      if (!hasDetectedChange) {
        console.warn(`No madlib changes detected after pivot switch. Waited full timeout (${timeout}ms). Proceeding anyway...`);
      } else {
        console.warn(`Madlib stability timeout reached (${timeout}ms). Proceeding anyway...`);
      }
    } catch (error) {
      console.warn('Error while waiting for madlib stability:', error);
      // Continue execution even if stability check fails
    }
  }

  /**
   * Helper method to wait for a specified duration
   * @param {number} ms - Milliseconds to wait
   */
  private async waitForTimeout(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }

  /**
   * Validates madlib text using regex pattern matching with detailed results
   * @param expectedPattern - The regex pattern to match against
   * @returns Promise<{isValid: boolean, actualText: string, expectedPattern: string, errorMessage?: string}> - Detailed validation result
   */
  public async validateMadlibWithPattern(expectedPattern: string): Promise<{
    isValid: boolean;
    actualText: string;
    expectedPattern: string;
    errorMessage?: string;
  }> {
    try {
      const actualText = await this.getMadlibText();
      const regex = new RegExp(expectedPattern, 'i'); // Case insensitive matching
      const isValid = regex.test(actualText);
      
      return {
        isValid,
        actualText,
        expectedPattern,
        errorMessage: isValid ? undefined : `Pattern "${expectedPattern}" did not match actual text: "${actualText}"`
      };
    } catch (error) {
      console.error('Failed to validate madlib with pattern:', error);
      const errorMessage = `Failed to validate madlib with pattern: ${error}`;
      return {
        isValid: false,
        actualText: '',
        expectedPattern,
        errorMessage
      };
    }
  }

  /**
   * Ensure all sub-menus are closed after assertions.
   * Pick one of random menu item which has shouldDisplay: true
   * And click on sub menu which has shouldDisplay: true to close any open sub-menus.
   * @param menuOptions 
   */
  private async ensureAllSubMenusClosed(menuOptions: AssertionMenuOption[]) {
    console.log('Ensuring all sub-menus are closed after assertions...');
    const filteredMenus = menuOptions.filter(menu => menu.shouldDisplay);
    // Here random index is picked to avoid always picking the first menu item
    // If there are more than 2 filtered menus, pick a random one, else pick the last one
    const randomIndex = filteredMenus.length > 2 ? randomInt(0, filteredMenus.length) : filteredMenus.length - 1;
    const menu: AssertionMenuOption = filteredMenus[randomIndex];
    if (menu) {
      const menuSelectorText = PeopleMenu[menu.name].selectorText;
      // Hover on the main menu item
      const menuElement = this.webActionHandler
        .getWebElement(
          this.resourceManager.getDynamicResource(
            menu.subMenu?.length ? "peopleDropdownItemWithSubmenus" : "peopleDropdownItemWithoutSubmenus",
            menuSelectorText,
            DS_COMPONENT_NAME
          )
        );
      await menuElement.hover();
      // Click on sub menu item if exists to close all sub-menus
      if (menu.subMenu && menu.subMenu.length) {
        const subMenu = menu.subMenu.find((menu: AssertionSubMenuOption) => menu.shouldDisplay); // Example, subMenu = { name: "BOUGHT_EACH_OF", shouldDisplay: true }
        if (subMenu && subMenu.name) {
          const subMenuSelector = PeopleMenu[subMenu.name].selectorText; // Example, selectorText = "Bought each of"
          const subMenuElement = this.webActionHandler
            .getWebElement(
              this.resourceManager.getDynamicResource(
                "peopleDropdownItemSubMenuItems",
                subMenuSelector,
                DS_COMPONENT_NAME
              )

          await subMenuElement.click();
        } else {
          throw new Error(`
            Unable to close all submenus for menu ${menu.name}, please check if at least one of submenu set with shouldDisplay: true or check UI menu
          `);
        }
      } else { // No sub-menu to close
        console.log(`No sub-menu to close, clicking on main menu ${menu.name}`);
        await menuElement.click();
      }
    }
  }

public async setDimensionSelection(
  selections: string[],
  searchReq = true
): Promise<void> {
  try {
    for (const selection of selections) {
      if (searchReq) {
        // Enter the selection text in the search field
        await this.enterTextInSearchField(selection);
      }
      // Wait for the selection viewport to be available in the DOM
      await this.webActionHandler
        .getWebElement(this.resourceManager.getResource("dsSelectionsViewPort", DS_COMPONENT_NAME))
        .waitFor();
      // Click the checkbox corresponding to the selection
      const selectionElement = this.webActionHandler.getWebElement(
        this.resourceManager.getDynamicResource(
          "checkboxLabel",
          selection,
          DS_COMPONENT_NAME
        )
      );
      await selectionElement.click();
    }
    // Click the 'Done' button to finalize the selection process
    await this.clickDone();
  } catch (error) {
    console.error("Error in setDimensionSelection:", error);
    throw new Error(`Failed to set dimension selection: ${error instanceof Error ? error.message : String(error)}`);
  }
}

public async setMadlibFolder(
  folderName: string,
  searchReq = true
): Promise<void> {
  try {
    if (searchReq) {
       // Get the search icon element
      const searchIcon = this.webActionHandler.getWebElement(
        this.resourceManager.getResource("searchIcon", DS_COMPONENT_NAME)
      );
    // Check if the search icon is visible
      const isVisible = await searchIcon.isVisible();
      if (isVisible) {
        // Click on the search icon to activate the search input
        await this.clickSearchIcon();
        // Enter the folder name into the search field
        await this.enterTextInSearchField(folderName);
      } else {
        // If the search icon is not visible, directly enter the folder name
        // Directly select the folder without searching
        await this.webActionHandler.delay(SMALL_TIMEOUT);
        await this.enterTextInSearchField(folderName);
      }
      // Select the folder from the search results
      await this.selectFolder(folderName);
    } else {
      // Search not required in new UI flows; select directly from visible list.
      await this.selectFolder(folderName);
    }
    console.log(`Folder '${folderName}' successfully selected.`);
  } catch (error) {
    console.error(
      `❌ Failed to select Madlib folder '${folderName}':`,
      error
    );
    throw error;
  }
}

5. HANDLING MULTIPLE MATCHING RESULTS
The component uses >> nth=0 to disambiguate when multiple items match:

From data-selector-resources.json:

"selectorValue": "div.item-value-checkbox-label >> text=${text} >> nth=0"
Navigation item selector (Line 363):
"selectorValue": "[data-test-id='ds-navigationItemLabel_${text}'] >> nth=0"
Mechanism:

>> nth=0 is a Playwright chaining operator that selects the first matching element (index 0)
If "APPLIANCES" appears in multiple category groupings, only the first occurrence in DOM order is clicked
This assumes the first occurrence is the desired one, or results are ordered such that the most relevant match comes first
Example:// If "APPLIANCES" appears in:
// <div class="item-value-checkbox-label">APPLIANCES</div>  // ← This one selected (nth=0)
// <div class="item-value-checkbox-label">APPLIANCES (Legacy)</div>
// <div class="item-value-checkbox-label">APPLIANCES-2024</div>

const selector = "div.item-value-checkbox-label >> text=APPLIANCES >> nth=0";
