public async setDimensionSelection(selections: string[], searchReq = true): Promise<void> {
  for (const selection of selections) {
    if (searchReq) {
      await this.enterTextInSearchField(selection);
      // NO delay here — proceeds immediately
    }
    // Wait for actual DOM state change, not fixed time
    await this.webActionHandler
      .getWebElement(this.resourceManager.getResource("dsSelectionsViewPort", DS_COMPONENT_NAME))
      .waitFor();  // Waits for visibility, not a fixed delay
    await selectionElement.click();
  }
}
