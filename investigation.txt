# surveyBuilder.page.ts — helper extraction audit

## Helpers added
- `getResource(name: string)` — wraps `this.resourceManager.getResource(name, SURVEY_BUILDER_PAGE_COMPONENT_NAME)`.
- `clickWhenVisible(resourceName, what, timeoutMs = ELEMENT_WAIT_TIMEOUT)` — get element via `getResource` → `waitUntilVisible` → `click()`.
- `fillWhenVisible(resourceName, text, what, timeoutMs = ELEMENT_WAIT_TIMEOUT)` — get element via `getResource` → `waitUntilVisible` → `enterText()`.

## Blocks replaced
- `getResource(...)` replaced ~25 standalone `this.resourceManager.getResource('x', SURVEY_BUILDER_PAGE_COMPONENT_NAME)` call sites (including ones feeding `isVisible`-only or `getWebElementsText` reads, and the ones later spread into dynamic `nth=` selectors).
- `clickWhenVisible(...)` replaced 15 "get → waitUntilVisible → click" blocks: `clickQuestionTypeResource`, `addOptionToQuestion`, `toggleRequired`, `clickBackButton`, `clickSelectPanelistsStep`, `clickAddQuestion`, `enterReviewMode` (first half), `deleteSelectedQuestions`, `selectAll`, `expandAll`, `collapseAll`, `unselectAll`, `exitReviewMode` (first half), `myAssetsTreeItem`/`savePendingChangesButton`/`saveDraftButton` clicks in `fillSaveCopyDialog`/`saveDraft`, `publishSurvey` (publishButton), `confirmPublish` (confirmButton), `setSurveyTitle` (editIcon).
- `fillWhenVisible(...)` replaced 3 "get → waitUntilVisible → enterText" blocks: `fillRatingQuestionText`, `setSurveyTitle` (titleInput), `fillSaveCopyDialog` (titleInput).

## Left as-is, and why
- `fillQuestionText` / `fillLabeledField` — same fill pattern but resolved via `getDynamicResource` (index/label-parameterized), which `fillWhenVisible` (built on the static-name `getResource`) doesn't support; not an exact match.
- `selectQuestionInReview` — click target resolved via `getDynamicResource('selectQuestionByIndex', …)`, same reason.
- `reopenCollapsedQuestion`, `getDuplicateQuestionElement`, `getQuestionRowElement`, `addRowToQuestion`/`addColumnToQuestion` — resource is spread into a custom `nth=`-scoped selector (or chained off another `WebElement`) before the click, not a plain `getResource` → click call.
- `hoverScaleAddMenuTrigger` — ends in `.hover()`, not `.click()`/`.enterText()`.
- `isAddQuestionTooltipVisibleOnHover` / `isDuplicateTooltipVisibleOnHover` — hover + visible/hidden tooltip polling, not a click/fill.
- `confirmPublishDialogText` / `publishingSurveyStatus` (in `confirmPublish`) and `titleInput`/`confirmDialogText` (in `publishSurvey`) — wait-then-read/branch, no click or fill follows, so only `getResource` applies.
- `waitForQuestionCountAtLeast` / `verifyPublishSucceeded` poll loops — no get+wait+click/fill shape at all.

Public method names, signatures, and return types are unchanged.

import { BasePage, WebActions } from '@gfk/gta';
import { CommonPage } from '../common/common-page';
import {
  ELEMENT_WAIT_TIMEOUT,
  UI_STATE_UPDATE_DELAY,
  PANEL_TRANSITION_DELAY,
  MAX_QUESTIONS_TOOLTIP_TIMEOUT,
  PUBLISH_SUCCESS_TOAST_TIMEOUT,
  QUESTION_TYPE_PICKER_TIMEOUT,
  QUESTION_TYPE_TILE_FIRST_CHECK_TIMEOUT,
  QUESTION_SAVE_TIMEOUT,
  PUBLISH_NAVIGATION_TIMEOUT,
  PUBLISH_TOAST_CHECK_TIMEOUT,
  PUBLISH_POST_CLICK_TIMEOUT
} from '../../constants/timeout-constants';
import { envConfig } from '../../config/env.config';
import surveyBuilderPageResources from '../../locators/survey/surveyBuilder.locator.json';
import devTestData from '../../test-data/dev-test-data.json';

const SURVEY_BUILDER_PAGE_COMPONENT_NAME = 'surveyBuilderPage';

export class SurveyBuilderPage extends CommonPage {
  constructor(basePage: BasePage) {
    super(basePage);
    this.resourceManager.registerResources(
      SURVEY_BUILDER_PAGE_COMPONENT_NAME,
      surveyBuilderPageResources
    );
  }

  /** Throws (unlike `waitFor()`, which does not throw on timeout in this framework) if `element` isn't visible within `timeoutMs`. */
  private async waitUntilVisible(
    element: ReturnType<WebActions['getWebElement']>,
    what: string,
    timeoutMs: number = ELEMENT_WAIT_TIMEOUT
  ): Promise<void> {
    if (!(await element.isVisible(timeoutMs))) {
      throw new Error(`${what} not visible after ${timeoutMs}ms`);
    }
  }

  private getResource(name: string) {
    return this.resourceManager.getResource(name, SURVEY_BUILDER_PAGE_COMPONENT_NAME);
  }

  /** Gets `resourceName` by its registered static name, waits for it to be visible, then clicks it. */
  private async clickWhenVisible(resourceName: string, what: string, timeoutMs: number = ELEMENT_WAIT_TIMEOUT): Promise<void> {
    const element = this.webActionHandler.getWebElement(this.getResource(resourceName));
    await this.waitUntilVisible(element, what, timeoutMs);
    await element.click();
  }

  /** Gets `resourceName` by its registered static name, waits for it to be visible, then fills it. */
  private async fillWhenVisible(resourceName: string, text: string, what: string, timeoutMs: number = ELEMENT_WAIT_TIMEOUT): Promise<void> {
    const element = this.webActionHandler.getWebElement(this.getResource(resourceName));
    await this.waitUntilVisible(element, what, timeoutMs);
    await element.enterText(text);
  }

  // Question type selection
  private async clickQuestionTypeResource(resourceName: string): Promise<void> {
    await this.clickWhenVisible(resourceName, `Question-type tile "${resourceName}"`, QUESTION_TYPE_PICKER_TIMEOUT);
  }

  // Shared field helpers (operate on the currently expanded/editing question)
  private async fillQuestionText(questionIndex: number, text: string): Promise<void> {
    const questionText = this.webActionHandler.getWebElement(
      this.resourceManager.getDynamicResource('questionTextInput', String(questionIndex), SURVEY_BUILDER_PAGE_COMPONENT_NAME)
    );
    await this.waitUntilVisible(questionText, 'Question text input');
    await questionText.enterText(text);
  }

  /** Fills Option N / Row N / Column N fields by their visible role label, e.g. 'Option 1', 'Row 2'. */
  private async fillLabeledField(label: string, value: string): Promise<void> {
    const field = this.webActionHandler.getWebElement(
      this.resourceManager.getDynamicResource('optionInputByLabel', label, SURVEY_BUILDER_PAGE_COMPONENT_NAME)
    );
    await this.waitUntilVisible(field, `Field "${label}"`);
    await field.enterText(value);
  }

  /** Fills Rating's question-text field — locator is unscoped by question index (interim `:visible`-based mitigation). */
  private async fillRatingQuestionText(text: string): Promise<void> {
    await this.fillWhenVisible('ratingQuestionTextInput', text, 'Rating question text input');
  }

  /**
   * Computes the questionIndex for a new question the same way the app itself does:
   * `surveyService.getQuestions().length + 1` (see question-editor.component.ts's
   * `upcomingQuestionNumber`). Counts the rendered `.question-number` markers on
   * already-saved questions rather than trusting any hardcoded literal, since the
   * persisted survey's question count is not stable across runs.
   */
  private async getNextQuestionIndex(): Promise<number> {
    const markers = await this.webActionHandler.getWebElementsText(this.getResource('questionNumberMarker'));
    return markers.length + 1;
  }

  /** Counts rendered `.question-number` markers, for TC_BS_REVIEW_MODE's delete-count assertion. */
  async getQuestionCount(): Promise<number> {
    try {
      const markers = await this.webActionHandler.getWebElementsText(this.getResource('questionNumberMarker'));
      return markers.length;
    } catch (error) {
      throw new Error(`Failed to get question count: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** Re-expands the question card after it collapses/saves on field-fill.
   *  `editQuestionButton` ([niqtooltip='Edit']) matches one icon per question, so once 2+ questions
   *  exist it's a Playwright strict-mode violation; scope to the LAST match (the question just added). */
  private async reopenCollapsedQuestion(): Promise<void> {
    const questionCount = await this.getQuestionCount();
    const editButtonResource = this.getResource('editQuestionButton');
    const editButton = this.webActionHandler.getWebElement({
      ...editButtonResource,
      selectorType: 'playwrights',
      selectorValue: `${editButtonResource.selectorValue} >> nth=${Math.max(questionCount - 1, 0)}`
    });
    await this.waitUntilVisible(editButton, 'Edit question button');
    await editButton.click();
  }

  async addOptionToQuestion(): Promise<void> {
    try {
      await this.clickWhenVisible('addOptionButton', 'Add option button');
    } catch (error) {
      throw new Error(`Failed to add option to question: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** Hovers the Add-menu trigger wrapper (`.matrix-action-area`) so the Row/Column buttons render — confirmed via source (scale-editor.component.html). */
  private async hoverScaleAddMenuTrigger() {
    const addMenuTrigger = this.webActionHandler.getWebElement(this.getResource('scaleAddMenuTrigger'));
    await this.waitUntilVisible(addMenuTrigger, 'Scale add-menu trigger');
    await addMenuTrigger.hover();
    return addMenuTrigger;
  }

  /** Shared by Point Scale and Multi Scale — both use the same survey-scale-editor matrix-actions mechanism. */
  async addRowToQuestion(): Promise<void> {
    try {
      const addMenuTrigger = await this.hoverScaleAddMenuTrigger();
      const addRow = addMenuTrigger.getWebElement(this.getResource('addRowButton'));
      await this.waitUntilVisible(addRow, 'Add row button');
      await addRow.click();
    } catch (error) {
      throw new Error(`Failed to add row to question: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** Shared by Point Scale and Multi Scale — both use the same survey-scale-editor matrix-actions mechanism. */
  async addColumnToQuestion(): Promise<void> {
    try {
      const addMenuTrigger = await this.hoverScaleAddMenuTrigger();
      const addColumn = addMenuTrigger.getWebElement(this.getResource('addColumnButton'));
      await this.waitUntilVisible(addColumn, 'Add column button');
      await addColumn.click();
    } catch (error) {
      throw new Error(`Failed to add column to question: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** `requiredToggle` ([data-test-id='-input-checkbox']) — confirmed correct per Codegen recording; the truncated id with no real prefix is how the live DOM renders it. */
  async toggleRequired(): Promise<void> {
    try {
      await this.clickWhenVisible('requiredToggle', 'Required toggle');
      await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
    } catch (error) {
      throw new Error(`Failed to toggle required: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** TC_BS_12 — footer's "Back" button (footer.component.html), no data-test-id, resolved by text. */
  async clickBackButton(): Promise<void> {
    try {
      await this.clickWhenVisible('backButton', 'Back button', 10000);
      await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
    } catch (error) {
      throw new Error(`Failed to click back button: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** TC_BS_12 — stepper's clickable circle (niq-step-circle, athena-core StepComponent), text doesn't navigate; index 0 = Select Panelists. */
  async clickSelectPanelistsStep(): Promise<void> {
    try {
      await this.clickWhenVisible('select-panelists-step-circle', 'Select Panelists step circle', 10000);
      await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
    } catch (error) {
      throw new Error(`Failed to click Select Panelists step: ${error instanceof Error ? error.message : error}`);
    }
  }

  async clickAddQuestion(): Promise<void> {
    try {
      await this.clickWhenVisible('addQuestionLink', 'Add question link');
      await this.webActionHandler.delay(PANEL_TRANSITION_DELAY);
    } catch (error) {
      throw new Error(`Failed to click add question: ${error instanceof Error ? error.message : error}`);
    }
  }

  // ---------------------------------------------------------------------
  // Per-question-type methods (TC_BS_03–TC_BS_09)
  // Each assumes it is called right after clickAddQuestion() (or for
  // question 1, right after landing on Build Survey), while this question
  // is the currently expanded one.
  // ---------------------------------------------------------------------

  /** TC_BS_03 */
  async addSingleChoiceQuestion(questionText: string, options: string[]): Promise<void> {
    try {
      const questionIndex = await this.getNextQuestionIndex();
      await this.clickQuestionTypeResource('questionTypeSingleChoice');
      await this.fillQuestionText(questionIndex, questionText);
      for (let i = 0; i < options.length; i++) {
        if (i >= 2) {
          await this.reopenCollapsedQuestion();
          await this.addOptionToQuestion();
        }
        await this.fillLabeledField(`Option ${i + 1}`, options[i]);
      }
    } catch (error) {
      throw new Error(`Failed to add Single Choice question: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** TC_BS_04 */
  async addMultipleChoiceQuestion(questionText: string, options: string[]): Promise<void> {
    try {

  /** Polls `getQuestionNumbers()` until its length reaches `target`, or throws after `timeout` ms. */
  private async waitForQuestionCountAtLeast(target: number, timeout: number): Promise<string[]> {
    const start = Date.now();
    let questionNumbers = await this.getQuestionNumbers();
    while (questionNumbers.length < target) {
      if (Date.now() - start >= timeout) {
        throw new Error(`Timed out waiting for question count to reach ${target} (current: ${questionNumbers.length})`);
      }
      await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
      questionNumbers = await this.getQuestionNumbers();
    }
    return questionNumbers;
  }

  /** TC_BS_PUBLISH/@ReviewMode precondition — tops up to `min` Rating questions, stopping as soon as the
   *  saved count (`.question-number`) reaches it. The Rating tile is a toggle (clicking add-question again
   *  while it's open closes it with ✕), so add-question is only clicked when the tile isn't already visible. */
  async ensureQuestionCount(min: number): Promise<void> {
    try {
      const { questionText } = devTestData.surveyQuestions.rating;
      let questionNumbers = await this.getQuestionNumbers();

      while (questionNumbers.length < min) {
        const countBeforeAdd = questionNumbers.length;
        const typeTile = this.webActionHandler.getWebElement(this.getResource('questionTypeRating'));

        if (!(await typeTile.isVisible(QUESTION_TYPE_TILE_FIRST_CHECK_TIMEOUT))) {
          await this.clickAddQuestion();
          if (!(await typeTile.isVisible(QUESTION_TYPE_PICKER_TIMEOUT))) {
            throw new Error(`Question-type picker did not open after clicking add-question (current question count: ${countBeforeAdd})`);
          }
        }

        await this.addRatingQuestion(`${questionText} ${countBeforeAdd + 1}`);
        questionNumbers = await this.waitForQuestionCountAtLeast(countBeforeAdd + 1, QUESTION_SAVE_TIMEOUT);
      }
    } catch (error) {
      throw new Error(`Failed to ensure question count: ${error instanceof Error ? error.message : error}`);
    }
  }

  /** TC_BS_PUBLISH — post-Confirm outcome check. Polls up to 60s: success once the URL has redirected to
   *  `/my-assets` and the success toast confirms it; fails fast as soon as `.survey-action-toast` shows
   *  anything other than the success text (survey-publish-popup.component.ts's error handler reuses the
   *  same `containerClass: 'survey-action-toast'` for `InsertToast('error', <backend message>, ...)`, so
   *  there is no separate error-only locator to check). */
  async verifyPublishSucceeded(confirmClickTime: number): Promise<boolean> {
    try {
      const elapsedSeconds = () => ((Date.now() - confirmClickTime) / 1000).toFixed(1);
      const actionToast = this.webActionHandler.getWebElement(this.getResource('survey-action-toast'));
      const successToast = this.webActionHandler.getWebElement(this.getResource('publish-success-toast'));

      const deadline = Date.now() + PUBLISH_NAVIGATION_TIMEOUT;
      while (Date.now() < deadline) {
        const currentUrl = await this.webActionHandler.getPageUrl();
        if (currentUrl.includes('my-assets')) {
          if (await successToast.isVisible(PUBLISH_TOAST_CHECK_TIMEOUT)) {
            return true;
          }
          const toastText = (await actionToast.isVisible()) ? await actionToast.getText() : 'none visible';
          throw new Error(`"Survey published successfully" toast did not appear after redirect (url: ${currentUrl}, elapsed: ${elapsedSeconds()}s, .survey-action-toast text: ${toastText})`);
        }

        if (await actionToast.isVisible(0)) {
          const toastText = await actionToast.getText();
          if (!toastText.includes('Survey published successfully')) {
            throw new Error(`Publish failed: "${toastText}" (url: ${currentUrl}, elapsed: ${elapsedSeconds()}s)`);
          }
        }

        await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
      }

      const currentUrl = await this.webActionHandler.getPageUrl();
      const toastText = (await actionToast.isVisible()) ? await actionToast.getText() : 'none visible';
      throw new Error(`Timed out waiting for navigation to /my-assets after Confirm (url: ${currentUrl}, elapsed: ${elapsedSeconds()}s, .survey-action-toast text: ${toastText})`);
    } catch (error) {
      throw new Error(`Failed to verify publish succeeded: ${error instanceof Error ? error.message : error}`);
    }
  }
