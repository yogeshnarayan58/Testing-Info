  // Question type selection
  private async clickQuestionTypeResource(resourceName: string): Promise<void> {
    const typeButton = this.webActionHandler.getWebElement(
      this.resourceManager.getResource(resourceName, SURVEY_BUILDER_PAGE_COMPONENT_NAME)
    );
    await typeButton.waitFor(QUESTION_TYPE_PICKER_TIMEOUT); // fail fast if the tile never renders, instead of hanging 5m
    await typeButton.click();
    await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
  }

  // Shared field helpers (operate on the currently expanded/editing question)
  private async fillQuestionText(questionIndex: number, text: string): Promise<void> {
    const questionText = this.webActionHandler.getWebElement(
      this.resourceManager.getDynamicResource('questionTextInput', String(questionIndex), SURVEY_BUILDER_PAGE_COMPONENT_NAME)
    );
    await questionText.waitFor();
    await questionText.enterText(text);
  }

  /** Fills Option N / Row N / Column N fields by their visible role label, e.g. 'Option 1', 'Row 2'. */

      this.resourceManager.getDynamicResource('optionInputByLabel', label, SURVEY_BUILDER_PAGE_COMPONENT_NAME)
    );
    await field.waitFor();
    await field.enterText(value);

    await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);

  /** Polls `getQuestionNumbers()` until its length reaches `target`, or throws after `timeout` ms. */
  private async waitForQuestionCountAtLeast(target: number, timeout: number): Promise<string[]> {
    const start = Date.now();
    let questionNumbers = await this.getQuestionNumbers();
    while (questionNumbers.length < target) {
      if (Date.now() - start >= timeout) {
        throw new Error(`Timed out waiting for question count to reach ${target} (current: ${questionNumbers.length})`);
      }
      await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
      questionNumbers = await this.getQuestionNumbers();
    }
    return questionNumbers;
  }

  /** TC_BS_PUBLISH/@ReviewMode precondition — tops up to `min` Rating questions. Per
   *  `build-survey.component.html` (`@if (questions().length === 0) { <survey-question-type-selector> }
   *  @else { <survey-preview-mode> }`), the `add-question` button only exists once
   *  `survey-preview-mode` has rendered (i.e. after question 1 is saved) — so question 1 is added
   *  directly. After each save, waits for the question count to actually increase, then confirms
   *  add-question is visible/enabled and the type tile really opens before adding the next one. */
  async ensureQuestionCount(min: number): Promise<void> {
    const { questionText } = devTestData.surveyQuestions.rating;
    let questionNumbers = await this.getQuestionNumbers();

    while (questionNumbers.length < min) {
      const countBeforeAdd = questionNumbers.length;
      await this.addRatingQuestion(`${questionText} ${countBeforeAdd + 1}`);
      questionNumbers = await this.waitForQuestionCountAtLeast(countBeforeAdd + 1, QUESTION_SAVE_TIMEOUT);

      if (questionNumbers.length >= min) break;

      const addQuestion = this.webActionHandler.getWebElement(
        this.resourceManager.getResource('addQuestionLink', SURVEY_BUILDER_PAGE_COMPONENT_NAME)
      );
      await addQuestion.waitFor(ADD_QUESTION_VISIBLE_TIMEOUT);
      if (!(await addQuestion.isEnabled())) {
        throw new Error(`add-question is not enabled after ${questionNumbers.length} saved questions`);
      }

      await this.clickAddQuestion();
      const typeTile = this.webActionHandler.getWebElement(
        this.resourceManager.getResource('questionTypeRating', SURVEY_BUILDER_PAGE_COMPONENT_NAME)
      );
      if (!(await typeTile.isVisible(QUESTION_TYPE_TILE_FIRST_CHECK_TIMEOUT))) {
        // Workaround for an unconfirmed root cause where the first click sometimes doesn't open the picker — retry once before failing.
        await this.clickAddQuestion();
        if (!(await typeTile.isVisible(QUESTION_TYPE_PICKER_TIMEOUT))) {
          throw new Error(`Question-type picker did not open after clicking add-question twice (current question count: ${questionNumbers.length})`);
        }
      }
    }
  }

  /** TC_BS_PUBLISH — post-Confirm success check: waits for the `/my-assets` redirect (confirmPublish calls
   *  router.navigateByUrl('/my-assets/survey')), then checks the success toast. Throws with current URL,
   *  elapsed time, and any `.survey-action-toast` text if either wait fails. */
  async verifyPublishSucceeded(confirmClickTime: number): Promise<boolean> {
    const elapsedSeconds = () => ((Date.now() - confirmClickTime) / 1000).toFixed(1);
    const describeFailure = async (reason: string) => {
      const currentUrl = await this.webActionHandler.getPageUrl();
      const actionToast = this.webActionHandler.getWebElement(
        this.resourceManager.getResource('survey-action-toast', SURVEY_BUILDER_PAGE_COMPONENT_NAME)
      );
      const toastText = (await actionToast.isVisible()) ? await actionToast.getText() : 'none visible';
      return `${reason} (url: ${currentUrl}, elapsed: ${elapsedSeconds()}s, .survey-action-toast text: ${toastText})`;
    };

    const navigationDeadline = Date.now() + PUBLISH_NAVIGATION_TIMEOUT;
    let currentUrl = await this.webActionHandler.getPageUrl();
    while (!currentUrl.includes('my-assets')) {
      if (Date.now() >= navigationDeadline) {
        throw new Error(await describeFailure('Timed out waiting for navigation to /my-assets after Confirm'));
      }
      await this.webActionHandler.delay(UI_STATE_UPDATE_DELAY);
      currentUrl = await this.webActionHandler.getPageUrl();
    }

    const toast = this.webActionHandler.getWebElement(
      this.resourceManager.getResource('publish-success-toast', SURVEY_BUILDER_PAGE_COMPONENT_NAME)
    );
    const isToastVisible = await toast.isVisible(PUBLISH_TOAST_CHECK_TIMEOUT);
    if (!isToastVisible) {
      throw new Error(await describeFailure('"Survey published successfully" toast did not appear after redirect'));
    }
    return true;
  }

  /** TC_BS_PUBLISH — asserts "Confirm publish?" prompt, clicks Confirm, and waits for the "Publishing survey" status. */
  async confirmPublish(): Promise<boolean> {
    try {
      const confirmPublishDialogText = this.webActionHandler.getWebElement(

  test('TC_BS_14: Review Mode select all, de-select one, delete, and exit keeps deletions', { tag: '@ReviewModeActions' }, async () => {
    const surveyBuilderPage = new SurveyBuilderPage(basePage);

    try {
      await surveyBuilderPage.collapseAll();
      await surveyBuilderPage.expandAll();
      const questionCountBeforeDelete = await surveyBuilderPage.getQuestionCount();
      await surveyBuilderPage.selectAll();

      for (let index = 0; index < questionCountBeforeDelete; index++) {
        assertTrue(await surveyBuilderPage.isQuestionSelected(index), `Expected checkbox ${index} to be ticked after Select All`);
      }

      await surveyBuilderPage.selectQuestionInReview(0);
      assertTrue(!(await surveyBuilderPage.isQuestionSelected(0)), 'Expected question 0 checkbox to clear after de-selecting it');
      for (let index = 1; index < questionCountBeforeDelete; index++) {
        assertTrue(await surveyBuilderPage.isQuestionSelected(index), `Expected checkbox ${index} to remain ticked after de-selecting question 0`);
      }

      await surveyBuilderPage.deleteSelectedQuestions();
      const deletedCount = questionCountBeforeDelete - 1;
      const questionCountAfterDelete = await surveyBuilderPage.getQuestionCount();
      assertTrue(
        questionCountAfterDelete === questionCountBeforeDelete - deletedCount,
        `Expected question count to drop by ${deletedCount}, found ${questionCountAfterDelete}`
      );

      const questionNumbersAfterDelete = await surveyBuilderPage.getQuestionNumbers();
      assertTrue(
        questionNumbersAfterDelete.every((number, idx) => number.trim() === `${idx + 1}.`),
        'Expected remaining questions to be renumbered sequentially'
      );

      await surveyBuilderPage.exitReviewMode();
      const questionCountAfterExit = await surveyBuilderPage.getQuestionCount();
      assertTrue(questionCountAfterExit === questionCountAfterDelete, 'Expected Build Survey view to keep deletions after exiting Review Mode');
    } catch (error) {
      throw new Error(`Failed to complete Review Mode actions flow: ${error instanceof Error ? error.message : error}`);
    }
  });
