+/** Shared by every `describe.serial` suite's `afterAll`: closes the browser (finalizing the .webm video,
+ *  same as `closeBrowser()` alone did before), then attaches the video to the report if one was recorded.
+ *  A missing video (e.g. video not enabled) is logged, not thrown - it must never fail the suite. */
+export async function closeBrowserAndAttachVideo(basePage: BasePage, testInfo: TestInfo, attachmentName: string): Promise<void> {
+  const page = (BasePage as any).page as import('playwright').Page | undefined; // gta exposes no public getter for the raw Page
+  const video = page?.video();
+  await basePage.getWebActions().closeBrowser(); // closes the context, which finalizes the .webm file
+  const videoPath = await video?.path().catch(() => undefined);
+  if (videoPath) {
+    await testInfo.attach(attachmentName, { path: videoPath, contentType: 'video/webm' });
+  } else {
+    console.log(`${attachmentName}: video path unavailable`);
+  }
